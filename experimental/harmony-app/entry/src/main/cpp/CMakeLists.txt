cmake_minimum_required(VERSION 3.16)
project(frameprobe LANGUAGES CXX)
set(HARMONY_DEPS "$ENV{HOME}/Maindome/harmony/build/deps-ohos-arm64/install" CACHE PATH "Native OHOS dependencies")
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs PATHS "${HARMONY_DEPS}/lib/cmake/opencv4" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
add_library(frameprobe SHARED frameprobe.cpp)
target_include_directories(frameprobe PRIVATE ${OpenCV_INCLUDE_DIRS})
target_include_directories(frameprobe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../harmony-core")
target_compile_features(frameprobe PRIVATE cxx_std_17)
target_compile_options(frameprobe PRIVATE -Wall -Wextra -Werror -Wno-unused-command-line-argument)
target_link_libraries(frameprobe PRIVATE ${OpenCV_LIBS})
target_link_libraries(frameprobe PRIVATE dl)
set(HARMONY_CORE_BUILD "$ENV{HOME}/Maindome/harmony/build/maacore-ohos-arm64/bin")
if(EXISTS "${HARMONY_CORE_BUILD}/libMaaCore.so")
  foreach(CORE_LIBRARY MaaCore MaaUtils)
    add_library(${CORE_LIBRARY} SHARED IMPORTED)
    set_target_properties(${CORE_LIBRARY} PROPERTIES IMPORTED_LOCATION "${HARMONY_CORE_BUILD}/lib${CORE_LIBRARY}.so")
    target_link_libraries(frameprobe PRIVATE ${CORE_LIBRARY})
  endforeach()
  add_library(harmony_onnxruntime SHARED IMPORTED)
  set_target_properties(harmony_onnxruntime PROPERTIES IMPORTED_LOCATION
    "$ENV{HOME}/Maindome/harmony/build/vcpkg-ohos/installed/maa-arm64-ohos/lib/libonnxruntime.so")
  target_link_libraries(frameprobe PRIVATE harmony_onnxruntime)
endif()
target_link_libraries(frameprobe PUBLIC libace_napi.z.so libnative_avscreen_capture.so
  libnative_media_core.so libnative_buffer.so)
